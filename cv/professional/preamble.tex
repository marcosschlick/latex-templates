% Packages for a PDFLaTeX-compatible resume on Overleaf.
\usepackage[left=0.72in,top=0.65in,right=0.72in,bottom=0.65in]{geometry}
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage[main=english,brazil,spanish]{babel}
\usepackage[default]{sourcesanspro}
\usepackage{xcolor}
\usepackage{hyperref}
\usepackage{titlesec}
\usepackage{enumitem}

% Soft black is for text, dark gray is for dates and stacks, and blue is for sections and links.
\definecolor{bodytextcolor}{HTML}{282828}
\definecolor{secondarytextcolor}{HTML}{6B6B6B}
\definecolor{primarycolor}{HTML}{324B64}

% Links are blue and have no surrounding boxes.
\hypersetup{
    colorlinks=true,
    urlcolor=primarycolor,
    linkcolor=primarycolor,
    pdfborder={0 0 0}
}

% General text, section, and spacing settings.
\color{bodytextcolor}
\linespread{1.0}
\setlength{\parindent}{0pt}

% Keeping the rule in a command avoids a titlesec parsing issue with \titlerule's optional thickness.
\newcommand{\sectionrule}{\color{primarycolor}\titlerule[0.4pt]}
\titleformat{\section}
    {\large\bfseries\color{primarycolor}}
    {}{0em}{}
    [\sectionrule]
\titlespacing*{\section}{0pt}{0.85em}{0.30em}

% \resumesection{\sectiontitle}
\newcommand{\resumesection}[1]{\section*{#1}}

% Localized section labels used by content.tex.
\input{sections}

% Keep bullet lists compact while leaving room between items.
\setlist[itemize]{
    label=\textbullet,
    leftmargin=1.2em,
    itemsep=0.12em,
    topsep=0.18em,
    parsep=0pt,
    partopsep=0pt
}

% Reusable building blocks used in content.tex.

% \resumelink{visible text}{URL}
\newcommand{\resumelink}[2]{%
    \href{#2}{#1}%
}

% \resumeheader{name}{role}{contact lines}
\newcommand{\resumeheader}[3]{%
    \begin{center}
        {\Huge\bfseries\color{primarycolor}#1\par}
        \vspace{0.32em}
        {\large\bfseries #2\par}
        \vspace{0.45em}
        {\normalsize #3\par}
    \end{center}
}

% \resumeentry{title}{organization}{period}
\newcommand{\resumeentry}[3]{%
    \noindent\textbf{\textcolor{bodytextcolor}{#1}}\hfill{\small\textcolor{secondarytextcolor}{#3}}\par
    {\textit{\textcolor{bodytextcolor}{#2}}}\par
}

% \resumeproject{name}{stack}{URL}
\newcommand{\resumeproject}[3]{%
    \noindent\href{#3}{\textbf{\textcolor{bodytextcolor}{#1}}}\par
    {\small\textcolor{secondarytextcolor}{#2}}\par
}
