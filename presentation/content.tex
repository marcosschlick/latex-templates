\subsection{A Tally of the Survivors}
\begin{CodeSlide}[subtitle={An imagined count of those who escaped}, note={The 75 percent figure is an imagined example, not a historical count.}]
        \begin{Verbatim}[fontsize=\footnotesize,numbers=left,numbersep=5pt]
def protected_fraction(p, N):
    if N <= 0 or p < 0 or p > N:
        raise ValueError("Bad input")
    return p / N

def show_report(p, N):
    fraction = protected_fraction(p, N)
    remaining = N - p
    print(f"Protected: {p} of {N}")
    print(f"Remaining: {remaining}")
    print(f"Protection rate: {fraction:.0%}")

saved, total = 75, 100
show_report(saved, total)
      \end{Verbatim}
\end{CodeSlide}
