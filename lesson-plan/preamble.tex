% The minipage gives paragraphs and lists identical top and bottom padding.
\newcommand{\dossierbox}[2]{%
  \par\vspace{0.5em}%
  \noindent\begin{tabular}{|p{\dimexpr\linewidth-2\tabcolsep-2\arrayrulewidth\relax}|}
  \hline
  \rowcolor{dossierlightblue}\rule{0pt}{1.45em}\textbf{#1}\\\hline
  \begin{minipage}[t]{\linewidth}\vspace{0.65em}#2\end{minipage}\\[-0.2em]
  \rule{0pt}{0.65em}\\\hline
  \end{tabular}%
  \par\vspace{0.15em}%
}
